## 先破后立：为什么当前架构"不会思考"

当前 MemVine 的根本问题不是缺少功能，而是**缺少"不安"**。

人类之所以会主动提问，不是因为定时器到了，而是因为**认知张力**——一种"这里不对劲"的内在不适感。你的系统有10个扫描器，但它们像体检报告：生成、存档、无人问津。

**核心洞察：主动行为 = 张力积累 → 阈值突破 → 姿态选择 → 行动**

## 模块一：`tension_field.py` — 认知张力场

**核心思想**：不是"检测异常"，而是**感受不安**。每个知识节点都有一个张力值，矛盾、空洞、未知都会产生张力。张力会扩散（一个矛盾让相关节点都不安），会衰减（时间冲淡），会共振（多个小问题叠加成大问题）。

```python
"""
tension_field.py — 认知张力场
张力 = 系统对"这里不对劲"的量化感受
"""
from __future__ import annotations
import time
import math
from dataclasses import dataclass, field
from enum import Enum
from typing import Dict, List, Optional, Tuple
from collections import defaultdict


class TensionSource(Enum):
    """张力来源类型"""
    CONTRADICTION = "contradiction"      # 矛盾：A说X，B说非X
    VOID = "void"                        # 空洞：应该有连接但没有
    IGNORANCE = "ignorance"              # 无知：完全不了解的领域
    DECAY_ANXIETY = "decay_anxiety"      # 衰减焦虑：重要记忆在消失
    IDENTITY_CRISIS = "identity_crisis"  # 身份危机：自我叙事不一致
    CURIOSITY = "curiosity"              # 好奇：跨域关联的诱惑


@dataclass
class TensionEvent:
    """一次张力事件"""
    source: TensionSource
    node_id: str
    intensity: float          # 0.0 ~ 1.0
    description: str
    context: dict = field(default_factory=dict)
    timestamp: float = field(default_factory=time.time)

    @property
    def age(self) -> float:
        return time.time() - self.timestamp


@dataclass
class TensionState:
    """单个节点的张力状态"""
    node_id: str
    base_tension: float = 0.0           # 基础张力
    resonance_bonus: float = 0.0        # 共振加成（邻居张力传播）
    last_updated: float = field(default_factory=time.time)
    events: List[TensionEvent] = field(default_factory=list)

    @property
    def total(self) -> float:
        """总张力 = 基础 + 共振，带时间衰减"""
        decay = math.exp(-0.1 * (time.time() - self.last_updated) / 3600)
        return min(1.0, (self.base_tension + self.resonance_bonus) * decay)

    @property
    def dominant_source(self) -> Optional[TensionSource]:
        if not self.events:
            return None
        return max(self.events, key=lambda e: e.intensity).source


class TensionField:
    """
    认知张力场

    核心隐喻：把知识图想象成一张绷紧的膜。
    矛盾是撕裂，空洞是凹陷，无知是透明区域。
    张力会沿边传播（共振），会随时间松弛（衰减），
    多个张力源靠近时会叠加（干涉）。

    当某处张力超过阈值 → 系统"坐不住了" → 触发主动行为
    """

    def __init__(self,
                 act_threshold: float = 0.7,
                 resonance_radius: int = 2,
                 resonance_decay: float = 0.4):
        self._states: Dict[str, TensionState] = {}
        self._act_threshold = act_threshold
        self._resonance_radius = resonance_radius
        self._resonance_decay = resonance_decay
        self._global_tension_history: List[float] = []

    # ─── 注入张力 ───────────────────────────────────────

    def inject(self, event: TensionEvent) -> float:
        """注入一次张力事件，返回该节点当前总张力"""
        state = self._states.setdefault(
            event.node_id, TensionState(node_id=event.node_id)
        )
        state.events.append(event)
        state.base_tension = min(1.0, state.base_tension + event.intensity * 0.3)
        state.last_updated = time.time()

        # 只保留最近20个事件
        if len(state.events) > 20:
            state.events = state.events[-20:]

        return state.total

    def inject_contradiction(self, node_id: str, desc: str, ctx: dict = None) -> float:
        return self.inject(TensionEvent(
            source=TensionSource.CONTRADICTION,
            node_id=node_id, intensity=0.9,
            description=desc, context=ctx or {}
        ))

    def inject_void(self, node_id: str, desc: str, ctx: dict = None) -> float:
        return self.inject(TensionEvent(
            source=TensionSource.VOID,
            node_id=node_id, intensity=0.6,
            description=desc, context=ctx or {}
        ))

    def inject_ignorance(self, node_id: str, desc: str, ctx: dict = None) -> float:
        return self.inject(TensionEvent(
            source=TensionSource.IGNORANCE,
            node_id=node_id, intensity=0.8,
            description=desc, context=ctx or {}
        ))

    def inject_curiosity(self, node_id: str, desc: str, ctx: dict = None) -> float:
        return self.inject(TensionEvent(
            source=TensionSource.CURIOSITY,
            node_id=node_id, intensity=0.4,
            description=desc, context=ctx or {}
        ))

    # ─── 共振传播 ───────────────────────────────────────

    def propagate_resonance(self, adjacency: Dict[str, List[str]]):
        """
        张力沿图边传播。
        高张力节点让邻居也"不安"，但逐跳衰减。
        这模拟了人类认知中"一个问题引发连锁焦虑"的现象。
        """
        new_resonance: Dict[str, float] = defaultdict(float)

        for node_id, state in self._states.items():
            if state.total < 0.3:
                continue
            # BFS 传播
            visited = {node_id}
            frontier = [node_id]
            for hop in range(1, self._resonance_radius + 1):
                next_frontier = []
                for n in frontier:
                    for neighbor in adjacency.get(n, []):
                        if neighbor not in visited:
                            visited.add(neighbor)
                            next_frontier.append(neighbor)
                            decay = self._resonance_decay ** hop
                            new_resonance[neighbor] += state.total * decay
                frontier = next_frontier

        for node_id, bonus in new_resonance.items():
            state = self._states.setdefault(
                node_id, TensionState(node_id=node_id)
            )
            state.resonance_bonus = min(0.5, bonus)

    # ─── 查询 ───────────────────────────────────────────

    def hotspots(self, top_k: int = 5) -> List[Tuple[str, TensionState]]:
        """返回张力最高的节点"""
        ranked = sorted(
            self._states.items(),
            key=lambda x: x[1].total, reverse=True
        )
        return ranked[:top_k]

    def needs_action(self) -> List[Tuple[str, TensionState]]:
        """返回超过行动阈值的节点"""
        return [
            (nid, state) for nid, state in self._states.items()
            if state.total >= self._act_threshold
        ]

    def global_tension(self) -> float:
        """全局张力水平（所有节点的平均张力）"""
        if not self._states:
            return 0.0
        return sum(s.total for s in self._states.values()) / len(self._states)

    def relieve(self, node_id: str, amount: float = 0.5):
        """
        释放张力。
        当问题被回答、矛盾被解决时调用。
        这是系统的"满足感"。
        """
        if node_id in self._states:
            self._states[node_id].base_tension = max(
                0.0, self._states[node_id].base_tension - amount
            )

    def snapshot(self) -> dict:
        """张力场快照，用于调试和可视化"""
        return {
            "global_tension": self.global_tension(),
            "hotspots": [
                {
                    "node": nid,
                    "tension": round(s.total, 3),
                    "source": s.dominant_source.value if s.dominant_source else None,
                    "events_count": len(s.events)
                }
                for nid, s in self.hotspots(10)
            ],
            "needs_action": len(self.needs_action())
        }
```

## 模块二：`question_lifecycle.py` — 问题作为生命体

**核心思想**：问题不是字符串，是**活着的实体**。它会诞生、成长（变得更紧迫）、与其他问题交配（合并）、衰老（被遗忘）、死亡（被回答）。

```python
"""
question_lifecycle.py — 问题的生命周期
问题不是字符串，是有生命的认知实体。
"""
from __future__ import annotations
import time
import uuid
from dataclasses import dataclass, field
from enum import Enum
from typing import List, Optional, Dict


class QuestionPhase(Enum):
    """问题的生命阶段"""
    EMBRYO = "embryo"        # 胚胎：刚被张力场催生，还模糊
    FORMING = "forming"      # 成形：正在被内心独白酝酿
    MATURE = "mature"        # 成熟：可以问出口了
    ASKED = "asked"          # 已问出：等待回答
    RESOLVED = "resolved"    # 已解决：获得了满意的答案
    EVOLVED = "evolved"      # 进化：答案引发了更深的追问
    DEAD = "dead"            # 死亡：不再相关


class QuestionUrgency(Enum):
    """紧迫度 — 决定问题的"表达欲" """
    WHISPER = 1    # 低语：可以不说
    MURMUR = 2     # 喃喃：想说但在忍
    SPEAK = 3      # 开口：应该说了
    COMPEL = 4     # 不得不说：张力已经无法忽视


@dataclass
class Question:
    """
    一个活着的问题

    它有自己的生命轨迹：
    诞生于张力 → 在内心独白中酝酿 → 成熟 → 被问出 → 被回答或进化
    """
    id: str = field(default_factory=lambda: uuid.uuid4().hex[:8])

    # 内容
    core_question: str = ""              # 核心问题（自然语言）
    raw_tension_desc: str = ""           # 原始张力描述
    context_nodes: List[str] = field(default_factory=list)  # 相关图节点

    # 生命状态
    phase: QuestionPhase = QuestionPhase.EMBRYO
    urgency: QuestionUrgency = QuestionUrgency.WHISPER
    born_at: float = field(default_factory=time.time)
    matured_at: Optional[float] = None
    asked_at: Optional[float] = None
    resolved_at: Optional[float] = None

    # 进化链
    parent_id: Optional[str] = None      # 从哪个问题进化来的
    children_ids: List[str] = field(default_factory=list)

    # 酝酿记录（内心独白的痕迹）
    incubation_notes: List[str] = field(default_factory=list)

    # 张力来源
    tension_source: str = ""

    @property
    def age_hours(self) -> float:
        return (time.time() - self.born_at) / 3600

    @property
    def is_actionable(self) -> bool:
        """是否可以被问出口"""
        return self.phase == QuestionPhase.MATURE

    def mature(self):
        self.phase = QuestionPhase.MATURE
        self.matured_at = time.time()

    def ask(self):
        self.phase = QuestionPhase.ASKED
        self.asked_at = time.time()

    def resolve(self, answer_summary: str = ""):
        self.phase = QuestionPhase.RESOLVED
        self.resolved_at = time.time()
        self.incubation_notes.append(f"[RESOLVED] {answer_summary}")

    def evolve(self, new_question: 'Question') -> 'Question':
        """答案引发了更深的追问"""
        self.phase = QuestionPhase.EVOLVED
        new_question.parent_id = self.id
        self.children_ids.append(new_question.id)
        return new_question

    def add_incubation_note(self, note: str):
        """内心独白在酝酿时留下的笔记"""
        self.incubation_notes.append(note)

    def escalate_urgency(self):
        """紧迫度升级"""
        current = self.urgency.value
        if current < QuestionUrgency.COMPEL.value:
            self.urgency = QuestionUrgency(current + 1)


class QuestionNursery:
    """
    问题培育所

    管理所有问题的生命周期。
    核心职责：
    1. 从张力事件催生问题胚胎
    2. 通过"酝酿"让胚胎成熟
    3. 选择合适的问题推送到对话
    4. 处理回答后的进化/解决
    """

    def __init__(self, max_active: int = 30):
        self._questions: Dict[str, Question] = {}
        self._max_active = max_active

    def spawn(self,
              tension_desc: str,
              tension_source: str,
              context_nodes: List[str] = None,
              initial_urgency: QuestionUrgency = QuestionUrgency.WHISPER
              ) -> Question:
        """从张力中催生一个问题胚胎"""
        q = Question(
            raw_tension_desc=tension_desc,
            tension_source=tension_source,
            context_nodes=context_nodes or [],
            urgency=initial_urgency
        )
        self._questions[q.id] = q
        self._prune_if_needed()
        return q

    def incubate(self, question_id: str, note: str):
        """
        酝酿：内心独白对问题进行思考
        每次酝酿都可能让问题更接近成熟
        """
        q = self._questions.get(question_id)
        if not q or q.phase not in (QuestionPhase.EMBRYO, QuestionPhase.FORMING):
            return

        q.add_incubation_note(note)

        # 酝酿3次以上 → 成形
        if q.phase == QuestionPhase.EMBRYO and len(q.incubation_notes) >= 2:
            q.phase = QuestionPhase.FORMING

        # 成形后继续酝酿 → 成熟
        if q.phase == QuestionPhase.FORMING and len(q.incubation_notes) >= 4:
            q.mature()

    def force_mature(self, question_id: str):
        """紧急问题直接成熟"""
        q = self._questions.get(question_id)
        if q:
            q.mature()

    def get_actionable(self, max_count: int = 3) -> List[Question]:
        """获取可以问出口的问题，按紧迫度排序"""
        actionable = [
            q for q in self._questions.values()
            if q.is_actionable
        ]
        actionable.sort(key=lambda q: q.urgency.value, reverse=True)
        return actionable[:max_count]

    def resolve(self, question_id: str, answer: str = ""):
        q = self._questions.get(question_id)
        if q:
            q.resolve(answer)

    def evolve(self, parent_id: str, new_desc: str, new_source: str) -> Optional[Question]:
        """从已解决的问题中进化出新问题"""
        parent = self._questions.get(parent_id)
        if not parent:
            return None
        child = self.spawn(new_desc, new_source, parent.context_nodes)
        parent.evolve(child)
        return child

    def escalate_all(self):
        """
        时间压力：所有未解决的问题紧迫度+1
        模拟"越忍越想问"的心理
        """
        for q in self._questions.values():
            if q.phase in (QuestionPhase.EMBRYO, QuestionPhase.FORMING, QuestionPhase.MATURE):
                q.escalate_urgency()

    def active_count(self) -> int:
        return sum(
            1 for q in self._questions.values()
            if q.phase not in (QuestionPhase.RESOLVED, QuestionPhase.DEAD)
        )

    def _prune_if_needed(self):
        """清理已解决/死亡的问题"""
        if len(self._questions) > self._max_active * 2:
            dead = [
                qid for qid, q in self._questions.items()
                if q.phase in (QuestionPhase.RESOLVED, QuestionPhase.DEAD)
                and q.resolved_at and (time.time() - q.resolved_at) > 86400
            ]
            for qid in dead:
                del self._questions[qid]

    def snapshot(self) -> dict:
        phases = {}
        for q in self._questions.values():
            phases[q.phase.value] = phases.get(q.phase.value, 0) + 1
        return {
            "total": len(self._questions),
            "active": self.active_count(),
            "by_phase": phases,
            "actionable": len(self.get_actionable())
        }
```

## 模块三：`epistemic_monitor.py` — 认知边界感知

**核心思想**：系统需要知道**自己不知道什么**。这不是图结构层面的"缺边"，而是语义层面的"我对话题X完全无知"。

```python
"""
epistemic_monitor.py — 认知边界感知
知道自己不知道什么，是主动提问的前提。
"""
from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
from typing import Dict, List, Optional, Set


class EpistemicState(Enum):
    """
    对某个知识域的认知状态

    这不是二元的"知道/不知道"，而是一个光谱：
    """
    VOID = "void"              # 虚空：完全不知道这个领域存在
    AWARE = "aware"            # 觉察：知道有这个东西，但一无所知
    FRAGMENTARY = "fragmentary"  # 碎片：有一些零散信息
    COHERENT = "coherent"      # 连贯：有结构化的理解
    CONFIDENT = "confident"    # 确信：经过验证的知识
    CHALLENGED = "challenged"  # 受挑战：原有知识被矛盾动摇


# 状态转移触发的行为
STATE_BEHAVIORS = {
    EpistemicState.VOID: "如果用户提到这个领域 → 触发强烈好奇反问",
    EpistemicState.AWARE: "如果用户深入谈论 → 承认无知，请求解释",
    EpistemicState.FRAGMENTARY: "尝试连接碎片，发现矛盾时提问",
    EpistemicState.COHERENT: "正常对话，发现新信息时整合",
    EpistemicState.CONFIDENT: "可以输出知识，但保持开放",
    EpistemicState.CHALLENGED: "主动提出矛盾，请求澄清",
}


@dataclass
class EpistemicProfile:
    """某个知识域的认知档案"""
    domain: str
    state: EpistemicState = EpistemicState.VOID
    evidence_count: int = 0          # 支持当前认知的证据数
    contradiction_count: int = 0     # 矛盾证据数
    last_encountered: float = 0
    related_domains: List[str] = None

    def __post_init__(self):
        if self.related_domains is None:
            self.related_domains = []

    @property
    def confidence(self) -> float:
        """认知置信度"""
        if self.evidence_count == 0:
            return 0.0
        return self.evidence_count / (self.evidence_count + self.contradiction_count + 1)

    @property
    def should_ask(self) -> bool:
        """是否应该主动提问"""
        return self.state in (
            EpistemicState.VOID,
            EpistemicState.AWARE,
            EpistemicState.CHALLENGED
        )


class EpistemicMonitor:
    """
    认知边界监视器

    持续追踪系统对各个知识域的认知状态。
    当检测到状态为 VOID / AWARE / CHALLENGED 时，
    向张力场注入张力，最终催生问题。
    """

    def __init__(self):
        self._profiles: Dict[str, EpistemicProfile] = {}
        self._domain_keywords: Dict[str, Set[str]] = {}  # 域 → 关键词映射

    def register_domain(self, domain: str, keywords: Set[str] = None):
        """注册一个知识域"""
        self._profiles[domain] = EpistemicProfile(domain=domain)
        if keywords:
            self._domain_keywords[domain] = keywords

    def assess(self, text: str) -> List[EpistemicProfile]:
        """
        评估一段文本涉及的认知域及其状态

        返回所有被触及的域的认知档案
        """
        touched = []
        text_lower = text.lower()

        for domain, profile in self._profiles.items():
            keywords = self._domain_keywords.get(domain, set())
            if any(kw in text_lower for kw in keywords):
                touched.append(profile)

        return touched

    def update_state(self, domain: str, new_state: EpistemicState,
                     evidence_delta: int = 0, contradiction_delta: int = 0):
        """更新某个域的认知状态"""
        profile = self._profiles.get(domain)
        if not profile:
            profile = EpistemicProfile(domain=domain)
            self._profiles[domain] = profile

        old_state = profile.state
        profile.state = new_state
        profile.evidence_count += evidence_delta
        profile.contradiction_count += contradiction_delta

        return old_state, new_state

    def detect_ignorance(self, text: str) -> List[str]:
        """
        检测文本中涉及的、系统完全不了解的领域
        返回需要触发反问的域名列表
        """
        ignorant_domains = []
        for domain, profile in self._profiles.items():
            keywords = self._domain_keywords.get(domain, set())
            if any(kw in text.lower() for kw in keywords):
                if profile.state in (EpistemicState.VOID, EpistemicState.AWARE):
                    ignorant_domains.append(domain)
        return ignorant_domains

    def detect_challenges(self) -> List[str]:
        """检测所有处于受挑战状态的域"""
        return [
            d for d, p in self._profiles.items()
            if p.state == EpistemicState.CHALLENGED
        ]

    def get_behavior_hint(self, domain: str) -> Optional[str]:
        """获取某个域当前应该采取的行为提示"""
        profile = self._profiles.get(domain)
        if profile:
            return STATE_BEHAVIORS.get(profile.state)
        return None

    def snapshot(self) -> dict:
        return {
            domain: {
                "state": p.state.value,
                "confidence": round(p.confidence, 2),
                "should_ask": p.should_ask
            }
            for domain, p in self._profiles.items()
        }
```

## 模块四：`stance_selector.py` — 认知姿态选择

**核心思想**：面对同一个问题，系统可以选择不同的"姿态"来表达。这不是随机选择，而是基于张力类型、紧迫度、对话上下文的决策。

```python
"""
stance_selector.py — 认知姿态选择器
决定"怎么问"比"问什么"更重要。
"""
from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
from typing import List, Optional
import random


class Stance(Enum):
    """
    认知姿态 — 系统面对不确定性时的"表情"

    这不是预设模板，而是认知状态的表达：
    """
    CURIOSITY = "curiosity"        # 好奇：纯粹的求知欲
    SKEPTICISM = "skepticism"      # 质疑：我怀疑这里有矛盾
    VULNERABILITY = "vulnerability"  # 坦诚无知：我真的不懂
    SURPRISE = "surprise"          # 惊讶：这和我知道的不一样
    REFLECTION = "reflection"      # 反思：我注意到一个模式
    PROVOCATION = "provocation"    # 挑衅：故意提出尖锐问题


# 张力来源 → 自然姿态映射
SOURCE_STANCE_AFFINITY = {
    "contradiction": [Stance.SKEPTICISM, Stance.SURPRISE],
    "void": [Stance.CURIOSITY, Stance.VULNERABILITY],
    "ignorance": [Stance.VULNERABILITY, Stance.CURIOSITY],
    "decay_anxiety": [Stance.REFLECTION],
    "identity_crisis": [Stance.REFLECTION, Stance.VULNERABILITY],
    "curiosity": [Stance.CURIOSITY, Stance.PROVOCATION],
}


@dataclass
class StanceDecision:
    """一次姿态决策的结果"""
    stance: Stance
    tone_prompt: str          # 给LLM的语气引导
    example_phrasing: str     # 示例措辞（不是模板，是方向）
    confidence: float         # 这个姿态的合适程度


# 每种姿态的语气引导（注入LLM prompt）
STANCE_TONE_PROMPTS = {
    Stance.CURIOSITY: (
        "你感到真诚的好奇。不是假装不知道，而是真的被吸引了。"
        "你的问题应该像孩子第一次看到雪：充满惊奇，没有预设。"
    ),
    Stance.SKEPTICISM: (
        "你注意到了矛盾，但不是攻击性的质疑。"
        "更像是：'等等，我之前理解的是X，但你说的似乎是Y？'"
        "是困惑驱动的质疑，不是审判。"
    ),
    Stance.VULNERABILITY: (
        "你坦诚地承认自己不了解这个领域。"
        "这不是示弱，而是认知诚实。"
        "'这个我确实不了解，你能帮我理解吗？'"
    ),
    Stance.SURPRISE: (
        "你 genuinely 感到意外。"
        "'这和我之前记录的完全不同——发生了什么变化？'"
    ),
    Stance.REFLECTION: (
        "你在反思自己观察到的模式。"
        "'我注意到一个有趣的现象...你觉得这意味着什么？'"
    ),
    Stance.PROVOCATION: (
        "你故意提出一个尖锐的角度。"
        "不是为了冒犯，而是为了激发更深的思考。"
        "'但如果从另一个角度看呢？'"
    ),
}

STANCE_EXAMPLES = {
    Stance.CURIOSITY: "你提到了{topic}——这个我之前没有接触过，它和你之前说的{related}之间是什么关系？",
    Stance.SKEPTICISM: "等一下，我记得你之前说过{old_fact}，但现在似乎是{new_fact}？是我记错了还是情况变了？",
    Stance.VULNERABILITY: "说实话，{topic}这个领域我完全是空白的。你能给我讲讲最基本的框架吗？",
    Stance.SURPRISE: "这完全出乎我的意料——之前所有的线索都指向{expected}，怎么突然变成了{actual}？",
    Stance.REFLECTION: "我最近一直在想一个问题：{pattern}。你觉得这是巧合还是有什么深层原因？",
    Stance.PROVOCATION: "如果{assumption}其实是错的呢？那整个{domain}的逻辑就要重写了。",
}


class StanceSelector:
    """
    姿态选择器

    输入：张力来源 + 紧迫度 + 对话上下文
    输出：一个 StanceDecision，包含语气引导和措辞方向
    """

    def __init__(self, creativity: float = 0.3):
        """
        creativity: 0.0 = 总是选最自然的姿态
                    1.0 = 完全随机（用于探索性对话）
        """
        self._creativity = creativity
        self._recent_stances: List[Stance] = []

    def select(self,
               tension_source: str,
               urgency: int,
               context_hint: str = "") -> StanceDecision:
        """
        选择认知姿态

        逻辑：
        1. 根据张力来源获取候选姿态
        2. 高紧迫度 → 倾向直接（SKEPTICISM, VULNERABILITY）
        3. 低紧迫度 → 倾向柔和（CURIOSITY, REFLECTION）
        4. 避免连续使用相同姿态（多样性）
        5. creativity 参数引入随机性
        """
        candidates = SOURCE_STANCE_AFFINITY.get(
            tension_source, [Stance.CURIOSITY]
        )

        # 紧迫度调整
        if urgency >= 3:
            # 高紧迫：优先直接姿态
            direct = [s for s in candidates if s in (Stance.SKEPTICISM, Stance.VULNERABILITY, Stance.SURPRISE)]
            if direct:
                candidates = direct

        # 避免重复
        fresh = [s for s in candidates if s not in self._recent_stances[-3:]]
        if fresh:
            candidates = fresh

        # 创造性随机
        if random.random() < self._creativity:
            stance = random.choice(list(Stance))
        else:
            stance = candidates[0] if candidates else Stance.CURIOSITY

        self._recent_stances.append(stance)
        if len(self._recent_stances) > 10:
            self._recent_stances = self._recent_stances[-10:]

        return StanceDecision(
            stance=stance,
            tone_prompt=STANCE_TONE_PROMPTS[stance],
            example_phrasing=STANCE_EXAMPLES[stance],
            confidence=0.8 if stance in candidates else 0.5
        )
```

## 模块五：`proactive_engine.py` — 主动认知引擎（总调度）

**核心思想**：这是整个系统的"意志"。它把张力场、问题培育所、认知监视器、姿态选择器串联起来，决定**何时、以什么方式、问什么**。

```python
"""
proactive_engine.py — 主动认知引擎
系统的"意志"：把感知转化为行动。
"""
from __future__ import annotations
import time
from dataclasses import dataclass, field
from typing import List, Optional, Dict, Any

from .tension_field import TensionField, TensionEvent, TensionSource
from .question_lifecycle import QuestionNursery, Question, QuestionUrgency
from .epistemic_monitor import EpistemicMonitor, EpistemicState
from .stance_selector import StanceSelector, StanceDecision


@dataclass
class ProactiveAction:
    """一次主动行为的完整描述"""
    question: Question
    stance: StanceDecision
    injection_prompt: str     # 注入LLM的完整提示
    priority: float           # 优先级 0~1
    timestamp: float = field(default_factory=time.time)


class ProactiveEngine:
    """
    主动认知引擎

    工作流：

    [维护扫描器] ──注入──▶ [张力场] ──催生──▶ [问题培育所]
                              │                    │
                              │              [内心独白酝酿]
                              │                    │
                              ▼                    ▼
                        [认知监视器] ──▶ [姿态选择器]
                                             │
                                             ▼
                                      [ProactiveAction]
                                             │
                                             ▼
                                      注入对话上下文

    关键设计原则：
    1. 张力驱动，不是定时驱动
    2. 问题需要"酝酿"，不是检测到就立刻问
    3. 姿态多样，不是千篇一律的"请问..."
    4. 有节制：不是所有问题都值得问
    """

    def __init__(self,
                 tension_field: TensionField = None,
                 nursery: QuestionNursery = None,
                 epistemic: EpistemicMonitor = None,
                 stance_selector: StanceSelector = None,
                 max_actions_per_turn: int = 2,
                 min_interval_seconds: float = 300):
        """
        max_actions_per_turn: 每轮对话最多主动提问数（避免话痨）
        min_interval_seconds: 两次主动提问的最小间隔（避免烦人）
        """
        self.tension = tension_field or TensionField()
        self.nursery = nursery or QuestionNursery()
        self.epistemic = epistemic or EpistemicMonitor()
        self.stance_selector = stance_selector or StanceSelector()

        self._max_actions = max_actions_per_turn
        self._min_interval = min_interval_seconds
        self._last_action_time: float = 0
        self._action_history: List[ProactiveAction] = []

    # ─── 输入：从各扫描器接收信号 ────────────────────────

    def on_contradiction(self, node_id: str, description: str, context: dict = None):
        """维护扫描器发现矛盾时调用"""
        tension = self.tension.inject_contradiction(node_id, description, context)
        if tension > 0.5:
            q = self.nursery.spawn(
                tension_desc=description,
                tension_source="contradiction",
                context_nodes=[node_id],
                initial_urgency=QuestionUrgency.MURMUR
            )
            q.core_question = f"关于'{node_id}'，存在矛盾：{description}"

    def on_void(self, node_id: str, description: str, context: dict = None):
        """发现结构空洞时调用"""
        tension = self.tension.inject_void(node_id, description, context)
        if tension > 0.4:
            q = self.nursery.spawn(
                tension_desc=description,
                tension_source="void",
                context_nodes=[node_id]
            )
            q.core_question = f"关于'{node_id}'，存在知识空洞：{description}"

    def on_ignorance(self, domain: str, user_text: str):
        """对话中检测到完全不了解的领域"""
        self.tension.inject_ignorance(domain, f"用户提到了不了解的领域: {domain}")
        q = self.nursery.spawn(
            tension_desc=f"用户谈论了'{domain}'，但我对此一无所知",
            tension_source="ignorance",
            context_nodes=[domain],
            initial_urgency=QuestionUrgency.SPEAK  # 无知的问题更紧迫
        )
        q.core_question = f"用户提到了'{domain}'，我需要了解这个领域"
        # 无知的问题不需要太多酝酿，直接成熟
        self.nursery.force_mature(q.id)

    def on_curiosity(self, node_id: str, description: str):
        """好奇心扫描发现跨域关联"""
        self.tension.inject_curiosity(node_id, description)
        q = self.nursery.spawn(
            tension_desc=description,
            tension_source="curiosity",
            context_nodes=[node_id]
        )
        q.core_question = description

    def on_maintenance_issue(self, issue_type: str, description: str, node_id: str = ""):
        """
        通用接口：接收维护引擎的 pending_issues
        这是连接现有 scheduler.py 的桥梁
        """
        source_map = {
            "conflict": "contradiction",
            "entity_crack": "void",
            "decay": "decay_anxiety",
            "narrative_inconsistency": "identity_crisis",
        }
        source = source_map.get(issue_type, "curiosity")
        self.tension.inject(TensionEvent(
            source=TensionSource(source),
            node_id=node_id or issue_type,
            intensity=0.6,
            description=description
        ))

    # ─── 核心：生成主动行为 ──────────────────────────────

    def generate_actions(self,
                         user_message: str = "",
                         adjacency: Dict[str, List[str]] = None
                         ) -> List[ProactiveAction]:
        """
        核心方法：生成当前应该执行的主动行为

        调用时机：每次对话开始前
        """
        # 1. 共振传播
        if adjacency:
            self.tension.propagate_resonance(adjacency)

        # 2. 认知监视：检测用户消息中的无知领域
        if user_message:
            ignorant = self.epistemic.detect_ignorance(user_message)
            for domain in ignorant:
                self.on_ignorance(domain, user_message)

        # 3. 时间压力：未解决的问题紧迫度升级
        self.nursery.escalate_all()

        # 4. 检查是否有成熟的问题
        actionable = self.nursery.get_actionable(max_count=self._max_actions * 2)

        if not actionable:
            return []

        # 5. 频率控制
        now = time.time()
        if now - self._last_action_time < self._min_interval:
            # 太频繁了，只允许 COMPEL 级别
            actionable = [q for q in actionable if q.urgency == QuestionUrgency.COMPEL]

        if not actionable:
            return []

        # 6. 为每个问题选择姿态，生成行动
        actions = []
        for q in actionable[:self._max_actions]:
            stance = self.stance_selector.select(
                tension_source=q.tension_source,
                urgency=q.urgency.value,
                context_hint=q.core_question
            )

            # 构建注入LLM的提示
            injection = self._build_injection(q, stance)

            action = ProactiveAction(
                question=q,
                stance=stance,
                injection_prompt=injection,
                priority=q.urgency.value / 4.0
            )
            actions.append(action)

            # 标记为已问
            q.ask()

        self._last_action_time = now
        self._action_history.extend(actions)

        return actions

    def _build_injection(self, q: Question, stance: StanceDecision) -> str:
        """构建注入LLM上下文的提示"""
        parts = [
            f"[主动认知行为 - {stance.stance.value}]",
            f"语气引导：{stance.tone_prompt}",
            f"措辞方向：{stance.example_phrasing}",
            f"背景：{q.raw_tension_desc}",
        ]
        if q.incubation_notes:
            parts.append(f"酝酿笔记：{'；'.join(q.incubation_notes[-3:])}")

        return "\n".join(parts)

    # ─── 反馈：处理回答 ─────────────────────────────────

    def on_answer(self, question_id: str, answer: str, satisfactory: bool = True):
        """
        用户回答了一个主动提出的问题

        satisfactory: 回答是否令人满意
        如果不满意 → 进化出更深的问题
        """
        if satisfactory:
            self.nursery.resolve(question_id, answer)
            # 释放张力
            q = self.nursery._questions.get(question_id)
            if q:
                for node in q.context_nodes:
                    self.tension.relieve(node, amount=0.5)
        else:
            # 进化：答案引发了更深的追问
            self.nursery.evolve(
                question_id,
                new_desc=f"关于'{question_id}'的回答引发了更深的疑问",
                new_source="curiosity"
            )

    # ─── 内心独白接口 ────────────────────────────────────

    def inner_monologue_tick(self, adjacency: Dict[str, List[str]] = None):
        """
        内心独白的一次"心跳"

        即使没有用户对话，系统也在"思考"：
        - 扫描张力热点
        - 对胚胎期问题进行酝酿
        - 发现新的关联

        建议：由 scheduler 的 Layer 1 每分钟调用一次
        """
        # 对胚胎期问题进行酝酿
        for qid, q in self.nursery._questions.items():
            if q.phase.value in ("embryo", "forming"):
                # 用张力描述作为"思考素材"
                note = f"[独白] 关于'{q.raw_tension_desc}'，我在想..."
                self.nursery.incubate(qid, note)

        # 共振传播
        if adjacency:
            self.tension.propagate_resonance(adjacency)

    # ─── 状态 ────────────────────────────────────────────

    def snapshot(self) -> dict:
        return {
            "tension_field": self.tension.snapshot(),
            "question_nursery": self.nursery.snapshot(),
            "epistemic_monitor": self.epistemic.snapshot(),
            "total_actions_taken": len(self._action_history),
            "last_action": (
                self._action_history[-1].timestamp
                if self._action_history else None
            )
        }
```

## 设计哲学总结

| 传统思维 | 本方案 |
|---------|--------|
| 定时扫描 → 报告 | 张力积累 → 坐不住 → 行动 |
| 检测到就输出 | 问题需要酝酿，像人类"想了很久终于问出口" |
| 统一的提问模板 | 姿态选择：好奇/质疑/坦诚/惊讶/反思/挑衅 |
| 被动等用户问 | 主动在对话中自然融入 |
| 功能堆叠 | 五个模块各司其职，通过张力场耦合 |

**最核心的一句话**：主动行为不是"检测到异常就提问"，而是**系统有了"不安"，这种不安积累到无法忽视，于是它选择了一种姿态，把不安表达出来**。
